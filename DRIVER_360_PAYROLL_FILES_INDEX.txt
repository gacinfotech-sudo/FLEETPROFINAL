   <SalaryCard driverId={driver._id} driverName={driver.name} />

STEP 2: Use Hook in Components
   const { data } = useDriverPayrollSummary(driverId);
   return <div>{formatCurrency(data?.data.currentMonth.net)}</div>;

STEP 3: Backend Event Emissions
   payrollEmitter.emitPaymentProcessed(tenantId, driverId, amount, 'bank_transfer');

STEP 4: Enable Real-time Updates (Optional)
   setupPayrollRealtimeUpdates(tenantId, driverId, (notification) => {...});

API ENDPOINTS CREATED
================================================================================

1. GET /api/drivers/:id/payroll-summary
   Purpose: Quick dashboard view
   Cache: 5 minutes
   Response: currentMonth, lastPayment, nextPayment, ytdEarnings, status

2. GET /api/drivers/:id/payroll-details
   Purpose: Full breakdown with history
   Cache: 5 minutes
   Response: All summary data + details + paymentHistory

3. GET /api/drivers/:id/360-with-payroll
   Purpose: Integrated driver 360 + payroll
   Cache: 5 minutes
   Response: driver360 + payroll + metadata
